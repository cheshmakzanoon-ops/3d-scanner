# This subdirectory is opt-in, host-only and never linked into realityforge.
if(ANDROID)
    message(FATAL_ERROR "Exact planar owner QA is not qualified for Android")
endif()
set(RF_QA_BOOST_ROOT "" CACHE PATH "Locally extracted, SHA256-verified Boost 1.86.0 root")
if(NOT EXISTS "${RF_QA_BOOST_ROOT}/boost/version.hpp")
    message(FATAL_ERROR "Set RF_QA_BOOST_ROOT to verified Boost 1.86.0; no download is performed")
endif()
add_library(rf_exact_planar_owner_qa STATIC exact_planar_owner.cpp)
target_include_directories(rf_exact_planar_owner_qa PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(rf_exact_planar_owner_qa SYSTEM PUBLIC "${RF_QA_BOOST_ROOT}")
# Fixed backends otherwise use dynamically allocated Karatsuba scratch.
target_compile_definitions(rf_exact_planar_owner_qa PUBLIC BOOST_MP_KARATSUBA_CUTOFF=4097)
target_compile_options(rf_exact_planar_owner_qa PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(rf_exact_planar_owner_cli cli.cpp)
target_link_libraries(rf_exact_planar_owner_cli PRIVATE rf_exact_planar_owner_qa)
add_executable(rf_exact_planar_owner_tests test.cpp)
target_link_libraries(rf_exact_planar_owner_tests PRIVATE rf_exact_planar_owner_qa)
foreach(target rf_exact_planar_owner_cli rf_exact_planar_owner_tests)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
if(RF_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(rf_exact_planar_owner_qa PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(rf_exact_planar_owner_qa PUBLIC -fsanitize=address,undefined)
endif()
add_test(NAME native_exact_planar_owner_qa COMMAND rf_exact_planar_owner_tests)
