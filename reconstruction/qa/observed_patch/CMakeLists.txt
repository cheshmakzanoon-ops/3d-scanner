cmake_minimum_required(VERSION 3.22)
project(observed_patch_qa LANGUAGES CXX)
if(ANDROID)
 message(FATAL_ERROR "Observed patch experiment is host-only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PATCH_SANITIZE "Sanitize isolated QA" OFF)
add_library(observed_patch STATIC patch.cpp)
target_compile_options(observed_patch PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
if(PATCH_SANITIZE)
 target_compile_options(observed_patch PUBLIC -fsanitize=address,undefined,float-cast-overflow -fno-omit-frame-pointer)
 target_link_options(observed_patch PUBLIC -fsanitize=address,undefined,float-cast-overflow)
endif()
add_executable(observed_patch_tests tests.cpp)
target_link_libraries(observed_patch_tests PRIVATE observed_patch)
target_compile_options(observed_patch_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
enable_testing()
add_test(NAME observed_patch_synthetic COMMAND observed_patch_tests)
