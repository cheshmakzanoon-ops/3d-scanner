cmake_minimum_required(VERSION 3.22)
project(RealityForgeFuzz LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "Use host Clang with compiler-rt/libFuzzer")
endif()
set(RF_SANITIZE ON CACHE BOOL "Native sanitizer instrumentation" FORCE)
if(NOT TARGET realityforge)
  add_subdirectory(../.. native)
endif()
target_compile_options(realityforge PRIVATE -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
target_link_options(realityforge PRIVATE -fsanitize=address,undefined)
foreach(kind numeric checkpoint exports)
  add_executable(rf_fuzz_${kind} ${kind}.cpp)
  target_link_libraries(rf_fuzz_${kind} PRIVATE realityforge)
  target_compile_features(rf_fuzz_${kind} PRIVATE cxx_std_20)
  target_compile_options(rf_fuzz_${kind} PRIVATE -Wall -Wextra -Werror -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
  target_link_options(rf_fuzz_${kind} PRIVATE -fsanitize=fuzzer,address,undefined)
endforeach()
